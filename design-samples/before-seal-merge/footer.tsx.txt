import { getTranslations } from "next-intl/server";
import { Link } from "@/i18n/navigation";
import { Logo } from "./logo";

export async function Footer() {
  const t = await getTranslations("common");
  const col = (title: string, links: [string, string][]) => (
    <div>
      <h2 className="text-sm font-semibold text-white">{title}</h2>
      <ul className="mt-4 space-y-3 text-sm">
        {links.map(([href, label]) => (
          <li key={href}>
            <Link href={href} className="text-navy-300 transition-colors hover:text-white">
              {label}
            </Link>
          </li>
        ))}
      </ul>
    </div>
  );
  return (
    <footer className="no-print mt-24 bg-navy-950 text-navy-300">
      <div className="container-page grid gap-12 py-16 md:grid-cols-[1.4fr_1fr_1fr_1fr]">
        <div>
          <Logo brand={t("brand")} tone="light" />
          <p className="mt-4 font-display text-lg text-white">{t("tagline")}</p>
          <p className="mt-3 max-w-sm text-sm leading-relaxed">{t("footer.blurb")}</p>
        </div>
        {col(t("footer.learn"), [
          ["/courses", t("nav.courses")],
          ["/companies", t("nav.companies")],
          ["/verify", t("nav.verify")],
        ])}
        {col(t("footer.organisation"), [
          ["/about", t("nav.about")],
          ["/contact", t("nav.contact")],
          ["/sign-in", t("nav.signIn")],
        ])}
        {col(t("footer.legal"), [
          ["/legal/privacy", t("footer.privacy")],
          ["/legal/terms", t("footer.terms")],
          ["/legal/refunds", t("footer.refunds")],
        ])}
      </div>
      <div className="border-t border-white/10">
        <div className="container-page flex flex-col gap-3 py-6 text-xs leading-relaxed md:flex-row md:items-center md:justify-between">
          <p>{t("footer.rights", { year: String(new Date().getFullYear()) })}</p>
          <p className="max-w-2xl">{t("footer.disclaimer")}</p>
        </div>
      </div>
    </footer>
  );
}
