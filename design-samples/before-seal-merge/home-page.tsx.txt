import type { Metadata } from "next";
import { getTranslations, setRequestLocale } from "next-intl/server";
import { ArrowRight, Building2, CalendarX2, Check, ShieldCheck, UserRound } from "lucide-react";
import { database } from "@/server/db/client";
import { listInstructors, listPublishedCourses, listUpcomingSessions } from "@/server/services/catalog";
import { pick } from "@/lib/format";
import { ButtonLink } from "@/components/ui/button";
import { Alert, Badge, EmptyState, SectionHeading } from "@/components/ui/primitives";
import { CourseCard } from "@/components/site/course-card";
import { HeroIllustration } from "@/components/site/hero-illustration";
import { SessionRow } from "@/components/site/session-row";

export const dynamic = "force-dynamic";

export async function generateMetadata({ params }: { params: Promise<{ locale: string }> }): Promise<Metadata> {
  const { locale } = await params;
  const t = await getTranslations({ locale, namespace: "home" });
  return { title: { absolute: `${t("meta.title")} · ${(await getTranslations({ locale, namespace: "common" }))("brand")}` } };
}

export default async function HomePage({ params }: { params: Promise<{ locale: string }> }) {
  const { locale } = await params;
  setRequestLocale(locale);
  const [t, tc, db] = await Promise.all([getTranslations("home"), getTranslations("common"), database()]);
  const [courses, sessions, instructors] = await Promise.all([listPublishedCourses(db), listUpcomingSessions(db), listInstructors(db)]);
  const courseById = new Map(courses.map((c) => [c.id, c]));
  const steps = t.raw("journey.steps") as { title: string; body: string }[];
  const faq = t.raw("faq.items") as { q: string; a: string }[];
  const points = t.raw("hero.points") as string[];

  return (
    <>
      {/* Hero */}
      <section className="relative overflow-hidden bg-navy-900 bg-weave text-white">
        <div className="container-page grid items-center gap-16 pb-28 pt-16 lg:grid-cols-[1.05fr_0.95fr] lg:gap-12 lg:pb-32 lg:pt-24">
          <div>
            <p className="animate-rise inline-flex items-center gap-2 rounded-full border border-white/15 bg-white/5 px-3.5 py-1.5 text-sm font-medium text-emerald-300">
              <ShieldCheck aria-hidden className="size-4" />
              {t("hero.eyebrow")}
            </p>
            <h1 className="animate-rise mt-6 text-4xl font-semibold leading-[1.15] [animation-delay:80ms] sm:text-5xl lg:text-6xl lg:leading-[1.12]">{t("hero.title")}</h1>
            <p className="animate-rise mt-6 max-w-xl text-lg leading-relaxed text-navy-100 [animation-delay:160ms]">{t("hero.lead")}</p>
            <div className="animate-rise mt-9 flex flex-col gap-3 [animation-delay:240ms] sm:flex-row">
              <ButtonLink href="/courses" size="lg">
                {tc("actions.exploreCourses")}
                <ArrowRight aria-hidden className="size-4.5 rtl:-scale-x-100" />
              </ButtonLink>
              <ButtonLink href="/companies" size="lg" variant="outline-light">
                {tc("actions.trainTeam")}
              </ButtonLink>
            </div>
            <ul className="animate-rise mt-10 grid gap-3 text-sm text-navy-100 [animation-delay:320ms] sm:grid-cols-3">
              {points.map((p) => (
                <li key={p} className="flex items-start gap-2">
                  <Check aria-hidden className="mt-0.5 size-4 shrink-0 text-emerald-300" />
                  {p}
                </li>
              ))}
            </ul>
          </div>
          <HeroIllustration labels={t.raw("hero.illustration")} />
        </div>
      </section>

      {/* Courses */}
      <section className="container-page py-20 sm:py-24" aria-labelledby="courses-h">
        <div id="courses-h">
          <SectionHeading eyebrow={t("courses.eyebrow")} title={t("courses.title")} lead={t("courses.lead")} />
        </div>
        <div className="mt-12 grid gap-6 md:grid-cols-2 lg:grid-cols-3">
          {courses.slice(0, 3).map((c) => (
            <CourseCard key={c.id} course={c} next={sessions.find((s) => s.courseId === c.id && s.seatsLeft > 0) ?? null} />
          ))}
        </div>
      </section>

      {/* Journey */}
      <section className="border-y border-line bg-paper-2/60 py-20 sm:py-24">
        <div className="container-page">
          <SectionHeading eyebrow={t("journey.eyebrow")} title={t("journey.title")} />
          <ol className="mt-12 grid gap-x-6 gap-y-10 sm:grid-cols-2 lg:grid-cols-5">
            {steps.map((s, i) => (
              <li key={s.title} className="relative">
                <div className="flex items-center gap-3">
                  <span className="grid size-10 shrink-0 place-items-center rounded-full bg-navy-900 font-display text-sm font-semibold text-emerald-300">{i + 1}</span>
                  {i < steps.length - 1 ? <span aria-hidden className="hidden h-px flex-1 bg-line-strong lg:block" /> : null}
                </div>
                <h3 className="mt-5 text-lg font-semibold text-navy-900">{s.title}</h3>
                <p className="mt-2 text-sm leading-relaxed text-muted">{s.body}</p>
              </li>
            ))}
          </ol>
        </div>
      </section>

      {/* Benefits */}
      <section className="container-page py-20 sm:py-24">
        <SectionHeading eyebrow={t("benefits.eyebrow")} title={t("benefits.title")} />
        <div className="mt-12 grid gap-6 lg:grid-cols-2">
          <div className="rounded-3xl border border-line bg-white p-8 shadow-soft sm:p-10">
            <span className="grid size-12 place-items-center rounded-xl bg-emerald-50 text-emerald-600">
              <UserRound aria-hidden className="size-6" />
            </span>
            <h3 className="mt-6 text-2xl font-semibold text-navy-900">{t("benefits.individuals.title")}</h3>
            <ul className="mt-6 space-y-4">
              {(t.raw("benefits.individuals.items") as string[]).map((b) => (
                <li key={b} className="flex gap-3 text-navy-800">
                  <Check aria-hidden className="mt-1 size-4.5 shrink-0 text-emerald-600" />
                  {b}
                </li>
              ))}
            </ul>
          </div>
          <div className="rounded-3xl bg-navy-900 bg-weave p-8 text-white shadow-lift sm:p-10">
            <span className="grid size-12 place-items-center rounded-xl bg-white/10 text-emerald-300">
              <Building2 aria-hidden className="size-6" />
            </span>
            <h3 className="mt-6 text-2xl font-semibold">{t("benefits.companies.title")}</h3>
            <ul className="mt-6 space-y-4">
              {(t.raw("benefits.companies.items") as string[]).map((b) => (
                <li key={b} className="flex gap-3 text-navy-100">
                  <Check aria-hidden className="mt-1 size-4.5 shrink-0 text-emerald-300" />
                  {b}
                </li>
              ))}
            </ul>
            <ButtonLink href="/companies" variant="outline-light" className="mt-8">
              {t("benefits.companies.cta")}
              <ArrowRight aria-hidden className="size-4 rtl:-scale-x-100" />
            </ButtonLink>
          </div>
        </div>
      </section>

      {/* Instructors */}
      <section className="border-y border-line bg-white py-20 sm:py-24">
        <div className="container-page grid gap-12 lg:grid-cols-[1fr_1.1fr] lg:items-start">
          <div>
            <SectionHeading eyebrow={t("instructors.eyebrow")} title={t("instructors.title")} lead={t("instructors.lead")} />
            <ul className="mt-8 space-y-3">
              {(t.raw("instructors.points") as string[]).map((p) => (
                <li key={p} className="flex gap-3 text-navy-800">
                  <ShieldCheck aria-hidden className="mt-0.5 size-5 shrink-0 text-emerald-600" />
                  {p}
                </li>
              ))}
            </ul>
          </div>
          <div className="space-y-4">
            {instructors.some((i) => i.isSample) ? <Alert tone="warning">{t("instructors.sampleNote")}</Alert> : null}
            <ul className="grid gap-4 sm:grid-cols-2">
              {instructors.slice(0, 4).map((i) => (
                <li key={i.id} className="rounded-2xl border border-line bg-paper p-5">
                  <div className="flex items-center gap-3">
                    <span aria-hidden className="grid size-11 place-items-center rounded-full bg-navy-900 font-display font-semibold text-emerald-300">
                      {pick(locale, i.nameEn, i.nameAr).trim().charAt(0)}
                    </span>
                    <div className="min-w-0">
                      <p className="font-semibold text-navy-900">{pick(locale, i.nameEn, i.nameAr)}</p>
                      <p className="text-xs text-muted">{pick(locale, i.titleEn, i.titleAr)}</p>
                    </div>
                  </div>
                  <div className="mt-4 flex flex-wrap gap-2">
                    {i.isSample ? <Badge tone="warning">{tc("sample")}</Badge> : null}
                    {i.credentials.some((c) => !c.verified) || i.credentials.length === 0 ? <Badge>{t("instructors.pending")}</Badge> : null}
                  </div>
                </li>
              ))}
            </ul>
          </div>
        </div>
      </section>

      {/* Upcoming sessions */}
      <section className="container-page py-20 sm:py-24">
        <div className="flex flex-col gap-6 sm:flex-row sm:items-end sm:justify-between">
          <SectionHeading eyebrow={t("sessions.eyebrow")} title={t("sessions.title")} lead={tc("time.zoneNote")} />
          <ButtonLink href="/courses" variant="light" className="self-start sm:self-auto">
            {tc("actions.viewAll")}
          </ButtonLink>
        </div>
        {sessions.length === 0 ? (
          <div className="mt-10">
            <EmptyState icon={<CalendarX2 aria-hidden className="size-6" />} title={t("sessions.empty.title")} action={<ButtonLink href="/contact" variant="light">{tc("nav.contact")}</ButtonLink>}>
              {t("sessions.empty.body")}
            </EmptyState>
          </div>
        ) : (
          <ul className="mt-10 space-y-3">
            {sessions.slice(0, 5).map((s) => {
              const c = courseById.get(s.courseId);
              return c ? <SessionRow key={s.id} session={s} course={c} /> : null;
            })}
          </ul>
        )}
      </section>

      {/* FAQ */}
      <section className="border-t border-line bg-paper-2/60 py-20 sm:py-24">
        <div className="container-page grid gap-10 lg:grid-cols-[0.8fr_1.2fr]">
          <SectionHeading eyebrow={t("faq.eyebrow")} title={t("faq.title")} />
          <div className="divide-y divide-line rounded-3xl border border-line bg-white px-6 shadow-soft sm:px-8">
            {faq.map((f) => (
              <details key={f.q} className="group py-5">
                <summary className="flex list-none items-start justify-between gap-4 font-semibold text-navy-900 [&::-webkit-details-marker]:hidden">
                  {f.q}
                  <span aria-hidden className="mt-1 grid size-6 shrink-0 place-items-center rounded-full border border-line-strong text-sm transition-transform group-open:rotate-45">+</span>
                </summary>
                <p className="mt-3 leading-relaxed text-muted">{f.a}</p>
              </details>
            ))}
          </div>
        </div>
      </section>

      {/* CTA */}
      <section className="container-page pt-20 sm:pt-24">
        <div className="rounded-[2rem] bg-navy-900 bg-weave px-6 py-14 text-center text-white shadow-lift sm:px-12 sm:py-20">
          <h2 className="text-3xl font-semibold sm:text-4xl">{t("cta.title")}</h2>
          <p className="mx-auto mt-4 max-w-xl text-lg text-navy-100">{t("cta.lead")}</p>
          <div className="mt-8 flex flex-col justify-center gap-3 sm:flex-row">
            <ButtonLink href="/courses" size="lg">{t("cta.primary")}</ButtonLink>
            <ButtonLink href="/companies" size="lg" variant="outline-light">{t("cta.secondary")}</ButtonLink>
          </div>
        </div>
      </section>
    </>
  );
}
